SELECT
  cur.IdCurso,
  cur.Nombre AS Curso,
	cur.TVEducativa AS Modalidad,
  cur.RecusadoProfesor AS DPP,
	COALESCE(mat.total_matriculados, 0) AS total_matriculados,
	COALESCE(inte.suscripciones_globales, 0) AS suscripciones_globales,
  COALESCE(inte.suscripciones_pendientes, 0) AS suscripciones_pendientes,
	prof.profesor,
  prof.profesor_email,
  prof.profesor_telefono,
  (SELECT c2.RequierePlanDeMejora
     FROM cogiti001.convocatorias c2
    WHERE c2.IdCurso = cur.IdCurso
      AND c2.Activo = 1
      AND c2.FechaInicioCurso <= CURDATE()
    ORDER BY c2.FechaInicioCurso DESC
    LIMIT 1) AS RPM,
  ult.ultima_fecha_inicio,
  TIMESTAMPDIFF(MONTH, ult.ultima_fecha_inicio, CURDATE()) AS meses_sin_impartir,
  act.ultima_actualizacion,
	act.ultimo_contacto_por
FROM cogiti001.cursos cur
JOIN (
     SELECT IdCurso, MAX(FechaInicioCurso) AS ultima_fecha_inicio
     FROM cogiti001.convocatorias
     WHERE FechaInicioCurso BETWEEN '2015-01-01' AND CURDATE()
     GROUP BY IdCurso
   ) ult ON ult.IdCurso = cur.IdCurso
LEFT JOIN (
     SELECT DISTINCT IdCurso
     FROM cogiti001.convocatorias
     WHERE Activo = 1
       AND FechaInicioCurso > CURDATE()
   ) fut ON fut.IdCurso = cur.IdCurso
LEFT JOIN (
     SELECT oa.IdCurso, oa.fecha AS ultima_actualizacion, oa.creado_por AS ultimo_contacto_por
     FROM appsmith.obs_actualizaciones oa
     JOIN (
          SELECT IdCurso, MAX(creado_en) AS ultimo
          FROM appsmith.obs_actualizaciones
          GROUP BY IdCurso
        ) u ON u.IdCurso = oa.IdCurso AND u.ultimo = oa.creado_en
   ) act ON act.IdCurso COLLATE utf8_unicode_ci = cur.IdCurso
LEFT JOIN (
     SELECT c.IdCurso, COUNT(*) AS total_matriculados
     FROM cogiti001.matriculas m
     JOIN cogiti001.convocatorias c ON c.IdConvocatoria = m.IdConvocatoria
     WHERE m.Baja = 0
       AND m.Precio > 0
     GROUP BY c.IdCurso
   ) mat ON mat.IdCurso = cur.IdCurso
LEFT JOIN (
     SELECT Interes AS IdCurso,
            COUNT(*) AS suscripciones_globales,
            SUM(CASE WHEN COALESCE(UltimoEnviado,'') = '' THEN 1 ELSE 0 END) AS suscripciones_pendientes
     FROM cogiti001.interesados
     GROUP BY Interes
   ) inte ON inte.IdCurso = cur.IdCurso
LEFT JOIN (
     SELECT c4.IdCurso,
            TRIM(CONCAT(u.Nombre, ' ', u.Apellido1, ' ', COALESCE(u.Apellido2, ''))) AS profesor,
            u.Email AS profesor_email,
            u.Telefono AS profesor_telefono
     FROM cogiti001.convocatorias c4
     JOIN cogiti001.profesoresconvocatoria pc ON pc.IdConvocatoria = c4.IdConvocatoria
     JOIN cogiti001.usuarios u ON u.IdUsuario = pc.IdUsuario
     JOIN (
          SELECT IdCurso, MAX(FechaInicioCurso) AS ultima
          FROM cogiti001.convocatorias
          WHERE Activo = 1 AND FechaInicioCurso <= CURDATE()
          GROUP BY IdCurso
        ) uc ON uc.IdCurso = c4.IdCurso AND uc.ultima = c4.FechaInicioCurso
     WHERE pc.EsPrincipal = 1
       AND c4.Activo = 1
   ) prof ON prof.IdCurso = cur.IdCurso
WHERE cur.Activo = 1
  AND COALESCE(fut.IdCurso, '') = ''
	AND EsGratuito = 0
HAVING DPP = 1
    OR rpm = 1
    OR ult.ultima_fecha_inicio < DATE_SUB(CURDATE(), INTERVAL 1 YEAR)
ORDER BY meses_sin_impartir DESC;
