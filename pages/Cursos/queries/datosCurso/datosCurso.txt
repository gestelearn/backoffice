SELECT
  cur.IdCurso,
  cur.Nombre AS nombre_curso,
  COUNT(DISTINCT c.IdConvocatoria) AS total_convocatorias,
  ROUND(AVG(e._MediaProfesores), 2) AS nota_profesor,
  ROUND(AVG(e._MediaContenidos), 2) AS nota_contenidos,
  ROUND(AVG(e._MediaCalidadPrecio), 2) AS nota_calidad_precio,
  ROUND(AVG(e._ValoracionMedia), 2) AS valoracion_media,
  COALESCE(SUM(e._NumRespondidas), 0) AS total_encuestados,
  COUNT(DISTINCT e.IdEncuesta) AS convocatorias_valoradas,
	SUM(CASE WHEN c.Activo = 1 THEN 1 ELSE 0 END) AS convocatorias_impartidas,
  SUM(CASE WHEN c.Activo = 0 THEN 1 ELSE 0 END) AS convocatorias_canceladas,
  ROUND(100 * SUM(CASE WHEN c.Activo = 1 THEN 1 ELSE 0 END) / COUNT(*), 1) AS ratio_apertura,
  (SELECT CONCAT(u.Nombre, ' ', u.Apellido1, ' ', COALESCE(u.Apellido2, ''))
     FROM cogiti001.convocatorias c2
     JOIN cogiti001.profesoresconvocatoria pc ON pc.IdConvocatoria = c2.IdConvocatoria
     JOIN cogiti001.usuarios u ON u.IdUsuario = pc.IdUsuario
    WHERE c2.IdCurso = cur.IdCurso AND c2.Activo = 1 AND pc.EsPrincipal = 1
    ORDER BY c2.FechaInicioCurso DESC LIMIT 1) AS profesor_nombre,
  (SELECT u.Email
     FROM cogiti001.convocatorias c2
     JOIN cogiti001.profesoresconvocatoria pc ON pc.IdConvocatoria = c2.IdConvocatoria
     JOIN cogiti001.usuarios u ON u.IdUsuario = pc.IdUsuario
    WHERE c2.IdCurso = cur.IdCurso AND c2.Activo = 1 AND pc.EsPrincipal = 1
    ORDER BY c2.FechaInicioCurso DESC LIMIT 1) AS profesor_email
FROM cogiti001.cursos cur
JOIN cogiti001.convocatorias c ON c.IdCurso = cur.IdCurso
LEFT JOIN cogiti001.encuestas e ON e.IdConvocatoria = c.IdConvocatoria
WHERE cur.IdCurso = '{{buscador.text}}'
GROUP BY cur.IdCurso, cur.Nombre;
