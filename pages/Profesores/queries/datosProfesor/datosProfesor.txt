SELECT
  u.IdUsuario,
  TRIM(CONCAT(u.Nombre, ' ', u.Apellido1, ' ', COALESCE(u.Apellido2, ''))) AS nombre_profesor,
  u.Email,
  u.Telefono,
  COUNT(DISTINCT c.IdCurso) AS total_cursos,
  ROUND(AVG(CASE WHEN e._NumRespondidas > 0 THEN e._MediaProfesores END), 2) AS nota_media
FROM cogiti001.usuarios u
JOIN cogiti001.profesoresconvocatoria pc ON pc.IdUsuario = u.IdUsuario
JOIN cogiti001.convocatorias c ON c.IdConvocatoria = pc.IdConvocatoria
LEFT JOIN cogiti001.encuestas e ON e.IdConvocatoria = c.IdConvocatoria
WHERE u.IdUsuario = {{ SelectProfesor.selectedOptionValue }}
  AND pc.EsPrincipal = 1
  AND c.Activo = 1
GROUP BY u.IdUsuario, u.Nombre, u.Apellido1, u.Apellido2, u.Email, u.Telefono;
